const express = require('express'); const app = express(); app.use(express.json()); // Чтобы сервер понимал JSON-данные // Наша база данных в оперативной памяти (после перезапуска сервера очистится) const users = []; const videos = []; // ========================================== // 1. СИСТЕМА АККАУНТОВ (РЕГИСТРАЦИЯ, ВХОД, ВЫХОД) // ========================================== // Регистрация: просто пишем ник и пароль app.post('/api/register', (req, res) => { const { username, password } = req.body; if (!username || !password) { return res.status(400).json({ error: "Нужно указать ник и пароль!" }); } const userExists = users.find(u => u.username.toLowerCase() === username.toLowerCase()); if (userExists) { return res.status(400).json({ error: "Этот никнейм уже занят!" }); } const newUser = { username, password, role: "user" }; users.push(newUser); res.status(201).json({ message: "Аккаунт успешно создан!", user: { username } }); }); // Вход в аккаунт: проверяем ник и пароль app.post('/api/login', (req, res) => { const { username, password } = req.body; const user = users.find(u => u.username.toLowerCase() === username.toLowerCase() && u.password === password); if (!user) { return res.status(401).json({ error: "Неверный ник или пароль!" }); } res.json({ message: "Вы успешно вошли!", username: user.username, role: user.role }); }); // Выход из аккаунта (На бэкенде мы просто подтверждаем действие, // а на самом сайте клиент стирает данные из памяти/cookies) app.post('/api/logout', (req, res) => { res.json({ message: "Вы успешно вышли из системы" }); }); // ========================================== // 2. ИИ-МОДЕРАЦИЯ И ЗАГРУЗКА ВИДЕО // ========================================== // Функция-модератор (ИИ-фильтр) function runAIModeration(title, description) { const text = (title + " " + description).toLowerCase(); // Список запрещенных слов для 18+ контента const nsfwBadWords = ["18+", "секс", "sex", "эротика", "пошлое", "порно"]; // Список триггеров для авторских прав (Copyright) const copyrightBadWords = ["disney", "sony music", "warner bros", "полный фильм", "слив трека"]; // Проверка на 18+ for (let word of nsfwBadWords) { if (text.includes(word)) { return { approved: false, reason: "БАН 18+: Обнаружен контент для взрослых или неприемлемые слова." }; } } // Проверка на Авторские Права for (let word of copyrightBadWords) { if (text.includes(word)) { return { approved: false, reason: "БАН АП: Обнаружено нарушение авторских прав (Content ID совпадение)." }; } } return { approved: true, reason: "" }; } // Загрузка нового ролика app.post('/api/videos/upload', (req, res) => { const { title, description, videoUrl, author } = req.body; if (!title || !videoUrl || !author) { return res.status(400).json({ error: "Название, ссылка на видео и автор обязательны!" }); } // Запускаем автоматическую ИИ модерацию const moderationResult = runAIModeration(title, description || ""); const newVideo = { id: "vid_" + Date.now(), title, description, videoUrl, author, views: 0, status: moderationResult.approved ? "approved" : "banned", banReason: moderationResult.reason }; videos.push(newVideo); if (newVideo.status === "banned") { return res.status(403).json({ message: "Видео загружено, но автоматически ЗАБАНЕНО модерацией!", video: newVideo }); } res.status(201).json({ message: "Видео успешно опубликовано!", video: newVideo }); }); // ========================================== // 3. ПРОСМОТР КОНТЕНТА И МОДЕРАЦИЯ // ========================================== // Получить список всех одобренных видео (для Главной страницы) app.get('/api/videos', (req, res) => { const approvedVideos = videos.filter(v => v.status === "approved"); res.json(approvedVideos); }); // Панель модератора: получить список вообще ВСЕХ забаненных роликов app.get('/api/admin/banned-videos', (req, res) => { const bannedVideos = videos.filter(v => v.status === "banned"); res.json(bannedVideos); }); // Администратор может вручную РАЗБАНИТЬ ролик app.post('/api/admin/unban/:id', (req, res) => { const videoId = req.params.id; const video = videos.find(v => v.id === videoId); if (!video) { return res.status(404).json({ error: "Видео не найдено" }); } video.status = "approved"; video.banReason = ""; res.json({ message: "Модератор одобрил видео вручную!", video }); }); // Запуск сервера на порту 3000 const PORT = 3000; app.listen(PORT, () => { console.log(`Сервер MyTube запущен на http://localhost:${PORT}`); console.log(`Скрипт готов к отправке на GitHub!`); });
